\section*{Mathematical notation}
\label{sec:notation-table}

\begin{table}[htbp]
\centering
\caption{Mathematical notation used in this research.}
\label{tab:notation}
\fitwidth{%
\begin{tabular}{lp{0.72\linewidth}}
\toprule
Symbol & Meaning \\
\midrule
$G_{\mathrm{approve}}$ & Directed endorsement graph from latest ERC-20 allowances (owner $\to$ spender) \\
$G_{\mathrm{transfer}}$ & Directed transfer graph (sender $\to$ receiver; AWP input) \\
$|V|$ & Number of nodes in a reputation graph \\
$|E|$ & Number of directed edges in a reputation graph \\
$w_{ij}$ & Weight of the directed edge from node $i$ to node $j$ \\
$o_i$ & Out-strength of node $i$, $o_i=\sum_j w_{ij}$ (written $o(u)$, or $o_A(u)$ and $o_T(u)$ per layer, in Chapter~\ref{ch:methodology}) \\
$P$ & Transition matrix, $P_{ij}=w_{ij}/o_i$ for $o_i>0$; a dangling row is uniform in Equation~\eqref{eq:transition} and zero in Equation~\eqref{eq:transition-impl}, where its mass is redistributed separately \\
$G_{\mathrm{PR}}$ & Google matrix $dP^{\top}+(1-d)\,\mathbf{1}\mathbf{1}^{\top}/|V|$ (Equation~\eqref{eq:google-matrix}) \\
$\mathbf{r}$, $r_i$ & PageRank vector and its entry for node $i$; $\mathbf{r}^{(t)}$ is the iterate after $t$ power-iteration steps \\
$\boldsymbol{\pi}$ & Teleportation (restart) distribution: uniform for EndorseRank and C-PR, proportional to activeness $X_u$ for AWP, and the normalized allowance-walk score for S-PR \\
$d$ & PageRank damping factor (default $0.85$); the teleport probability is $1-d$ \\
$\varepsilon$ & Convergence tolerance of power iteration ($10^{-8}$, on the $\|\cdot\|_1$ change between iterates) \\
$I_{\max}$ & Iteration cap of power iteration ($300$) \\
$\lambda$ & Allowance-layer weight of the coupled operator C-PR ($\lambda=1$ and $\lambda=0$: one layer walked alone) \\
$P^{(\lambda)}$ & Coupled transition matrix of C-PR with allowance-layer weight $\lambda$ (Equation~\eqref{eq:coupled-transition}) \\
$\alpha_A(u)$, $\alpha_T(u)$ & Weights of the allowance row and the transfer row in the coupled transition row of node $u$ \\
$k$ & Logistic time-decay steepness \\
$t_0$ & Logistic time-decay midpoint (days) \\
$\Delta t$ & Age in days of a transfer or approval \\
$\sigma(\Delta t)$ & Logistic time weight of a transfer (AWP) or latest approval (EndorseRank) \\
$V(z)$ & Bounded value weight of AWP and EndorseRank on a US-dollar amount $z$, $2/(1+e^{-bz})-1$ with $b=\ln 3/m$, so that the median amount $m$ ($m_T=200$ dollars for transfers, $m_A=62$ for allowances) gets $V=1/2$ \\
$U$ & Cap on one allowance in the allowance layer of C-PR and in the in-approve value ($20{,}000$ dollars) \\
$X_u$ & Activeness of address $u$, AWP's restart weight \\
$n$ & Number of contracts in a correlation (matched cohort $n=15{,}107$; spender cohorts $n=17{,}050$ in W0 and $18{,}152$ in W1) \\
$\rho$ & Spearman's rank correlation coefficient \\
$\tau$ & Kendall's rank correlation coefficient; every reported value is $\tau_b$ \\
$T_{\mathrm{obs}}$ & End of the observation window and score freeze of W1 (30~June~2026, 23:59:59 UTC) \\
$t_1$, $t_2$ & Score freeze of W0 (31~March~2026, 23:59:59 UTC) and end of its label window (30~June~2026) \\
$\mathcal{W}$ & Set of contracts in the evaluation cohort \\
$s(w)$ & Reputation score of contract $w$ under a given method \\
$p(w)$ & Proxy value for contract $w$ \\
\bottomrule
\end{tabular}
}
\end{table}
